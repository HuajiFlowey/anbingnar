#anbi_daxugo_harem_menu_action_type ：当前行动种类（1=marriage 2=stability 3=building 4=special），种类入口按钮 set_variable。
#变量直接覆盖写，无需清空辅助效果（原 clr_selected_flag / clr_action_type_flag 已随 flag 机制移除）。

#========== 行动脚本（2026-08-29 参数化重构） ==========
#调用链：action_list_<种类>_<N> 按钮 effect → anbi_daxugo_harem_menu_action_<种类>_<N>（按选中宫殿变量 8 路参数分发）
#  → 通用分发 anbi_daxugo_harem_menu_palace_action（变量守卫 + 钩子调用 + 置「本轮已执行」标记）
#  → 效果钩子 anbi_daxugo_harem_menu_action_<种类>_<N>_<宫殿>_effect（具体效果在此填充）
#冷却/周期规则（2026-09-12 改版）：
#  · marriage_1：仍走专属分发 <宫殿>_marriage_1_cooldown（duration 3650），它同时是「本轮周期门」——
#    冷却进行中 = 周期开启（婚姻 2~8 与稳定 1~8 各可执行一次）；冷却结束 = 周期关闭（仅婚姻行动一可用）。
#  · marriage_1 执行时：先 clr 本宫殿 marriage 2~8 与 stability 1~8 的 _used 标记（重置），再加冷却（开门）。
#  · marriage 2~8 / stability 1~8：无冷却，置 <宫殿>_{marriage,stability}_<N>_used，随婚姻行动一重置。
#  · building 1~8：无冷却，置 <宫殿>_building_<N>_used，永不清除 → 全程仅一次，不受周期影响。
#  · special：无冷却无标记（本次未改动）。
#  · 旧 <宫殿>_{marriage,stability,building}_cooldown 修饰符及其本地化已于 2026-09-12 删除；
#    仅 <宫殿>_marriage_1_cooldown 8 个保留（它们是周期门本体，删不得）。
#tooltip 分工（2026-09-12 起）：
#  · 按钮 tooltip 的「风味文本」→ _tt 段（tooltip = anbi_daxugo_harem_menu_action_list_<种类>_<N>_tt，
#    值 [Root.DaxugoHaremActionTT<N>] → ANBI_DAXUGO_HAREM_action_tt_<种类>_<N>_<宫殿>，Y91），当前是占位符。
#  · 按钮 tooltip 的「执行效果说明」→ 由本文件各 _effect 钩子内的 custom_tooltip 承载；
#    由人工在填充实际效果时自行添加（键名自定，本工作区不预置键）。marriage_1 ×8 钩子已有 custom_tooltip 可作样例。

#---- 通用宫殿行动分发（参数：$kind$=种类 $n$=按钮序号 $palace$=宫殿 $palace_value$=宫殿变量值） ----
#通用分发（无冷却）：钩子执行后置「已执行」标记；是否随周期重置由标记名与 marriage_1 的 clr 清单决定
anbi_daxugo_harem_menu_palace_action = {
	if = {
		limit = { is_variable_equal = { which = anbi_daxugo_harem_menu_selected value = $palace_value$ } }
		anbi_daxugo_harem_menu_action_$kind$_$n$_$palace$_effect = yes
		#无冷却（2026-09-12 改版）：不再加冷却修饰符，改为置「已执行」标记，可用性由条件模块的 NOT has_country_flag 控制
		#  marriage 2~8 / stability 1~8 → 由婚姻行动一 clr_country_flag 重置（本轮各一次）
		#  building 1~8               → 永不清除（全程仅一次，不受婚姻周期影响）
		set_country_flag = anbi_daxugo_harem_$palace$_$kind$_$n$_used
	}
}
#无冷却·无标记版（2026-09-12 起无人调用，保留备用）
anbi_daxugo_harem_menu_palace_action_nocd = {
	if = {
		limit = { is_variable_equal = { which = anbi_daxugo_harem_menu_selected value = $palace_value$ } }
		anbi_daxugo_harem_menu_action_$kind$_$n$_$palace$_effect = yes
	}
}

#marriage_1 专属独立冷却版：钩子执行后进入 <宫殿>_marriage_1 冷却（与其他行动的冷却互不影响）
anbi_daxugo_harem_menu_palace_action_marriage_1 = {
	if = {
		limit = { is_variable_equal = { which = anbi_daxugo_harem_menu_selected value = $palace_value$ } }
		anbi_daxugo_harem_menu_action_marriage_1_$palace$_effect = yes
		#重置后宫行动：清除本宫殿「本轮已执行」标记（婚姻 2~8 与稳定 1~8）
		#注：building 标记刻意不清——建筑行动为全程一次，与周期脱钩
		clr_country_flag = anbi_daxugo_harem_$palace$_marriage_2_used
		clr_country_flag = anbi_daxugo_harem_$palace$_marriage_3_used
		clr_country_flag = anbi_daxugo_harem_$palace$_marriage_4_used
		clr_country_flag = anbi_daxugo_harem_$palace$_marriage_5_used
		clr_country_flag = anbi_daxugo_harem_$palace$_marriage_6_used
		clr_country_flag = anbi_daxugo_harem_$palace$_marriage_7_used
		clr_country_flag = anbi_daxugo_harem_$palace$_marriage_8_used
		clr_country_flag = anbi_daxugo_harem_$palace$_stability_1_used
		clr_country_flag = anbi_daxugo_harem_$palace$_stability_2_used
		clr_country_flag = anbi_daxugo_harem_$palace$_stability_3_used
		clr_country_flag = anbi_daxugo_harem_$palace$_stability_4_used
		clr_country_flag = anbi_daxugo_harem_$palace$_stability_5_used
		clr_country_flag = anbi_daxugo_harem_$palace$_stability_6_used
		clr_country_flag = anbi_daxugo_harem_$palace$_stability_7_used
		clr_country_flag = anbi_daxugo_harem_$palace$_stability_8_used
		#开启本轮周期：该冷却本身就是周期门（冷却中 = 婚姻 2~8 / 稳定 1~8 可各执行一次）
		#冷却：marriage_1 专属（时长可调）
		add_country_modifier = {
			name = anbi_daxugo_harem_$palace$_marriage_1_cooldown
			duration = 3650
		}
	}
}
