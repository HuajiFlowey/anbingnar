#	custom_trigger_tooltip = {
#		tooltip = <本地化键>
#		<被包装的触发器...>
#	}

#参数：$kind$=行动种类(marriage/stability/building)  $n$=按钮序号  $palace$=宫殿名  $palace_value$=宫殿变量值
#变量只命中一个宫殿值，按钮侧 OR×8 调用后等价于「当前选中宫殿的可用条件模块」。
#special 不绑定宫殿，按钮 trigger 直连 anbi_daxugo_harem_menu_action_special_<N>_available，不经此触发器。

	is_variable_equal = { which = anbi_daxugo_harem_menu_selected value = $palace_value$ }
	anbi_daxugo_harem_menu_action_$kind$_$n$_$palace$_available = yes
}
